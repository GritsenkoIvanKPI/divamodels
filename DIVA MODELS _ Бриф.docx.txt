Где заложены эти коэффициенты в скрипте:
В JavaScript-коде за значения коэффициентов отвечают два объекта:
1. expMap (Коэффициент уровня опыта):
   * 0 (НОВИЧОК): 0.20 (20%)
   * 1 (РАСТУЩИЙ): 0.25 (25%)
   * 2 (СТАБИЛЬНЫЙ): 0.30 (30%)
   * 3 (ТОП-МОДЕЛЬ): 0.35 (35%)
2. tabooMap (Модификатор табу):
   * 0 (МНОГО): 0.1 (понижающий множитель 10%)
   * 1 (СТАНДАРТ): 1.0 (без изменений 100%)
   * 2 (БЕЗ ТАБУ): 1.1 (повышающий множитель 110%)

JavaScript
// Калькулятор
const followers = document.getElementById('followers');
const price = document.getElementById('price');
const expSlider = document.getElementById('exp-slider');
const tabooSlider = document.getElementById('taboo-slider');


const fVal = document.getElementById('f-val');
const pVal = document.getElementById('p-val');
const expLabel = document.getElementById('exp-label');
const tabooLabel = document.getElementById('taboo-label');
const resDisp = document.getElementById('calc-result');


const expMap = { 0: 0.20, 1: 0.25, 2: 0.30, 3: 0.35 };
const expNames = { 
    0: { ru: "НОВИЧОК", ua: "НОВАЧОК" }, 
    1: { ru: "РАСТУЩИЙ", ua: "ЗРОСТАЮЧИЙ" }, 
    2: { ru: "СТАБИЛЬНЫЙ", ua: "СТАБІЛЬНИЙ" }, 
    3: { ru: "ТОП-МОДЕЛЬ", ua: "ТОП-МОДЕЛЬ" } 
};


const tabooMap = { 0: 0.1, 1: 1.0, 2: 1.1 };
const tabooNames = { 
    0: { ru: "МНОГО", ua: "БАГАТО" }, 
    1: { ru: "СТАНДАРТ", ua: "СТАНДАРТ" }, 
    2: { ru: "БЕЗ ТАБУ", ua: "БЕЗ ТАБУ" } 
};


function calculate() {
    const lang = localStorage.getItem('diva_lang') || 'ru';
    const f = parseInt(followers.value);
    const p = parseFloat(price.value);
    
    expLabel.innerText = expNames[expSlider.value][lang];
    tabooLabel.innerText = tabooNames[tabooSlider.value][lang];
    
    fVal.innerText = f.toLocaleString();
    pVal.innerText = p;
    
    const totalBase = f * p;
    const currentPercent = expMap[expSlider.value];
    const tabooMod = tabooMap[tabooSlider.value];
    
    const min = Math.round(totalBase * currentPercent * tabooMod);
    const max = Math.round(min * 1.10);
    
    resDisp.innerText = `$${min.toLocaleString()} - $${max.toLocaleString()}`;
}
